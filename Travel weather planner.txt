has_bike = True
has_car = False
has_ride_share_app = True
is_raining = False
distance_mi = 3

if not distance_mi:
    print(False)

elif distance_mi <= 1 and not is_raining:
    print(True)

elif distance_mi > 1 and distance_mi <= 6:
    if has_bike and not is_raining:
        print(True)
    else:
        print(False)

else:
    if has_car or has_ride_share_app:
        print(True)
    else:
        print(False)
