has_bike = bool(input('Type True or False'))
has_car = bool(input('Type True or False'))
has_ride_share_app = bool(input('Type True or False'))
is_raining = bool(input('Type True or False'))
distance_mi = int(input('Type the amount of miles'))

if not distance_mi:
    print(False)

elif distance_mi <= 1 and not is_raining:
    print(True)

elif distance_mi > 1 and distance_mi <= 6:
    if has_bike and not is_raining:
        print(True)
    else:
        print(False)

else:
    if has_car or has_ride_share_app:
        print(True)
    else:
        print(False)
